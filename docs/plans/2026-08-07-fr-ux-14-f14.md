  // 보드/백로그 카드의 라벨 칩 행 — 최대 3개 + "+N" 오버플로 (FR-UX-14 F14)
  const MAX_VISIBLE_LABELS = 3
  export function CardLabelChips({ labels }: { labels: readonly string[] }): JSX.Element | null {
    if (labels.length === 0) return null                    // FR9 — 빈 div 도 만들지 않는다
    const visible = labels.slice(0, MAX_VISIBLE_LABELS)
    const hiddenCount = labels.length - visible.length
    const hidden = labels.slice(MAX_VISIBLE_LABELS)
    return (
      <div className="flex flex-wrap items-center gap-1">
        {visible.map((l) => (
          <Badge key={l} variant="neutral" className="max-w-[10rem] truncate text-muted-foreground">
            {l}
          </Badge>
        ))}
        {hidden.length > 0 && (
          <Badge
            variant="neutral"
            className="text-muted-foreground"
            aria-label={cardLabels.moreLabelsAriaLabel(hidden)}   // D6 — 터치엔 hover 가 없다
            title={hidden.join(', ')}                              // 마우스 보조 (주 경로 아님)
          >
            {`+${hidden.length}`}
          </Badge>
        )}
      </div>
    )
  }
