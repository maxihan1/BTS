```kotlin
/** 활성 댓글 단건 조회. issueId 까지 대조해 다른 이슈의 commentId 를 404 로 만든다(E2). */
fun findActive(id: UUID, issueId: UUID): Comment? =
    dsl.selectFrom(COMMENTS)
        .where(COMMENTS.ID.eq(id))
        .and(COMMENTS.ISSUE_ID.eq(issueId))
        .and(COMMENTS.DELETED_AT.isNull)
        .fetchOne()
        ?.let(::toComment)

fun updateBody(id: UUID, body: String, updatedAt: Instant): Int =
    dsl.update(COMMENTS)
        .set(COMMENTS.BODY, body)
        .set(COMMENTS.UPDATED_AT, updatedAt.toOffsetDateTime())
        .where(COMMENTS.ID.eq(id).and(COMMENTS.DELETED_AT.isNull))
        .execute()

fun softDelete(id: UUID, deletedAt: Instant): Int =
    dsl.update(COMMENTS)
        .set(COMMENTS.DELETED_AT, deletedAt.toOffsetDateTime())
        .where(COMMENTS.ID.eq(id).and(COMMENTS.DELETED_AT.isNull))
        .execute()
```

```kotlin
fun delete(actor: ActorId, issueKey: IssueKey, commentId: UUID) {
    checkPermission(actor, issueKey, IssuePermission.UPDATE)   // 공통 전제 — 던져도 됨
    archiveGuard.checkByIssue(issueKey)

    val issue = issueRepository.findByKey(issueKey) ?: throw IssueNotFoundException(issueKey)
    val existing = commentRepository.findActive(commentId, issue.id.value)
        ?: throw CommentNotFoundException(commentId)

    // FR-4: 작성자 OR 모더레이터. hasPermission 은 질의형(던지지 않음) — 여기서 던지면 작성자 경로가 죽는다.
    val isAuthor = existing.authorId == actor.value
    val isModerator = permissionResolver.hasPermission(
        actor.value, IssuePermission.SOFT_DELETE, IssueScope.Issue(issueKey.value),
    )
    if (!isAuthor && !isModerator) {
        throw IssueAccessDeniedException(actor, IssuePermission.SOFT_DELETE, IssueScope.Issue(issueKey.value))
    }

    if (commentRepository.softDelete(commentId, Instant.now(clock)) == 0) {
        throw CommentNotFoundException(commentId)
    }
    log.info("comment_deleted issueKey={} commentId={} actor={} moderated={}",
        issueKey.value, commentId, actor.value, !isAuthor)
}
```
